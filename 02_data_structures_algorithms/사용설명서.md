```python
class UserDatabase:
    def __init__(self):
        self.bst = BST()

    def add_user(self, user):
        self.bst.insert(user)

    def find_user(self, user_id):
        # BST에서 O(log n) 검색
        return self.bst.search(user_id)

    def get_users_range(self, start_id, end_id):
        # 범위 내 사용자 조회
        result = []
        def range_search(node):
            if not node:
                return
            if start_id < node.value.user_id:
                range_search(node.left)
            if start_id <= node.value.user_id <= end_id:
                result.append(node.value)
            if node.value.user_id < end_id:
                range_search(node.right)

        range_search(self.bst.root)
        return result
```

```python
class ScoreBoard:
    def __init__(self):
        self.bst = BST()

    def add_score(self, player, score):
        self.bst.insert((score, player))

    def get_top_n(self, n=10):
        # 역순으로 정렬 (높은 점수부터)
        all_scores = self.bst.inorder()
        return sorted(all_scores, reverse=True)[:n]

    def get_rank(self, score):
        # 해당 점수보다 높은 점수의 개수 + 1
        count = 0
        def count_higher(node):
            nonlocal count
            if not node:
                return
            if node.value[0] > score:
                count += 1
            count_higher(node.left)
            count_higher(node.right)

        count_higher(self.bst.root)
        return count + 1

# 사용
scoreboard = ScoreBoard()
scoreboard.add_score("플레이어1", 95)
scoreboard.add_score("플레이어2", 87)
scoreboard.add_score("플레이어3", 92)

print("Top 3:", scoreboard.get_top_n(3))
print("점수 90의 순위:", scoreboard.get_rank(90))
```

```python
class SocialNetwork:
    def __init__(self):
        self.graph = {}

    def add_friendship(self, user1, user2):
        if user1 not in self.graph:
            self.graph[user1] = []
        if user2 not in self.graph:
            self.graph[user2] = []

        self.graph[user1].append(user2)
        self.graph[user2].append(user1)

    def find_mutual_friends(self, user1, user2):
        """공통 친구 찾기"""
        friends1 = set(self.graph.get(user1, []))
        friends2 = set(self.graph.get(user2, []))
        return friends1 & friends2

    def friend_recommendations(self, user):
        """친구 추천 (친구의 친구)"""
        friends = set(self.graph.get(user, []))
        recommendations = set()

        for friend in friends:
            for friend_of_friend in self.graph.get(friend, []):
                if friend_of_friend != user and friend_of_friend not in friends:
                    recommendations.add(friend_of_friend)

        return recommendations

    def shortest_connection(self, user1, user2):
        """두 사용자 간 최단 연결 경로 (BFS)"""
        if user1 not in self.graph or user2 not in self.graph:
            return None

        queue = [(user1, [user1])]
        visited = {user1}

        while queue:
            current, path = queue.pop(0)

            if current == user2:
                return path

            for neighbor in self.graph.get(current, []):
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append((neighbor, path + [neighbor]))

        return None

# 사용 예제
sn = SocialNetwork()
sn.add_friendship("Alice", "Bob")
sn.add_friendship("Alice", "Charlie")
sn.add_friendship("Bob", "David")
sn.add_friendship("Charlie", "David")
sn.add_friendship("David", "Eve")

# 공통 친구
mutuals = sn.find_mutual_friends("Alice", "David")
print("공통 친구:", mutuals)

# 친구 추천
recommendations = sn.friend_recommendations("Alice")
print("추천 친구:", recommendations)

# 연결 경로
```

```python
class CityMap:
    def __init__(self):
        self.graph = {}

    def add_road(self, city1, city2, distance):
        if city1 not in self.graph:
            self.graph[city1] = {}
        if city2 not in self.graph:
            self.graph[city2] = {}

        self.graph[city1][city2] = distance
        self.graph[city2][city1] = distance

    def dijkstra(self, start, end):
        """최단 거리 경로 찾기"""
        import heapq

        distances = {node: float('inf') for node in self.graph}
        distances[start] = 0
        previous = {}
        queue = [(0, start)]

        while queue:
            current_distance, current_node = heapq.heappop(queue)

            if current_node == end:
                # 경로 재구성
                path = []
                while current_node:
                    path.append(current_node)
                    current_node = previous.get(current_node)
                return path[::-1], distances[end]

            if current_distance > distances[current_node]:
                continue

            for neighbor, weight in self.graph[current_node].items():
                distance = current_distance + weight

                if distance < distances[neighbor]:
                    distances[neighbor] = distance
                    previous[neighbor] = current_node
                    heapq.heappush(queue, (distance, neighbor))

        return None, float('inf')

# 사용
city_map = CityMap()
city_map.add_road("서울", "수원", 30)
city_map.add_road("서울", "인천", 40)
city_map.add_road("수원", "대전", 80)
city_map.add_road("인천", "대전", 120)
city_map.add_road("대전", "부산", 200)

path, distance = city_map.dijkstra("서울", "부산")
print(f"최단 경로: {' → '.join(path)}")
print(f"총 거리: {distance}km")
```

#### 작업 의존성 관리 (위상 정렬)
```python
class TaskScheduler:
    def __init__(self):
        self.tasks = {}
        self.dependencies = {}

    def add_task(self, task, depends_on=None):
        self.tasks[task] = True
        if depends_on:
            if task not in self.dependencies:
                self.dependencies[task] = []
            self.dependencies[task].extend(depends_on)

    def topological_sort(self):
        """작업 실행 순서 결정"""
        in_degree = {task: 0 for task in self.tasks}

        # 진입 차수 계산
        for task, deps in self.dependencies.items():
            for dep in deps:
                in_degree[dep] = in_degree.get(dep, 0) + 1

        # 진입 차수가 0인 작업부터 시작
        queue = [task for task, degree in in_degree.items() if degree == 0]
        result = []

        while queue:
            current = queue.pop(0)
            result.append(current)

            # 의존하는 작업의 진입 차수 감소
            for task, deps in self.dependencies.items():
                if current in deps:
                    in_degree[task] -= 1
                    if in_degree[task] == 0:
                        queue.append(task)

        return result

# 사용
scheduler = TaskScheduler()
scheduler.add_task("코딩")
scheduler.add_task("테스트", ["코딩"])
scheduler.add_task("배포", ["테스트"])
scheduler.add_task("문서화", ["코딩"])
scheduler.add_task("리뷰", ["문서화", "테스트"])

order = scheduler.topological_sort()
print("작업 순서:", " → ".join(order))
```

```python
    if n <= 1:
        return n

def fib_dp(n, memo={}):
    if n in memo:
        return memo[n]
    if n <= 1:
        return n

    memo[n] = fib_dp(n-1, memo) + fib_dp(n-2, memo)
    return memo[n]
```

#### 배낭 문제 (Knapsack)
```python
def knapsack(weights, values, capacity):
    """
    0-1 배낭 문제
    weights: 물건의 무게 리스트
    values: 물건의 가치 리스트
    capacity: 배낭 용량
    """
    n = len(weights)
    dp = [[0] * (capacity + 1) for _ in range(n + 1)]

    for i in range(1, n + 1):
        for w in range(1, capacity + 1):
            if weights[i-1] <= w:
                # 현재 물건을 넣을지 말지 선택
                dp[i][w] = max(
                    values[i-1] + dp[i-1][w - weights[i-1]],  # 넣는 경우
                    dp[i-1][w]  # 안 넣는 경우
                )
            else:
                dp[i][w] = dp[i-1][w]

    return dp[n][capacity]

# 쇼핑 최적화
items = [
    {"name": "노트북", "weight": 3, "value": 2000},
    {"name": "태블릿", "weight": 2, "value": 1500},
    {"name": "카메라", "weight": 1, "value": 1000},
    {"name": "책", "weight": 1, "value": 300}
]

weights = [item["weight"] for item in items]
values = [item["value"] for item in items]
capacity = 5  # 배낭 용량

max_value = knapsack(weights, values, capacity)
print(f"최대 가치: {max_value}")
```

#### 최장 공통 부분 수열 (LCS)

```python
def lcs(text1, text2):
    """최장 공통 부분 수열의 길이"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])

    return dp[m][n]

def lcs_string(text1, text2):
    """실제 LCS 문자열 반환"""
    m, n = len(text1), len(text2)
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    # DP 테이블 채우기
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if text1[i-1] == text2[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])

    # 역추적하여 문자열 구성
    i, j = m, n
    lcs = []

    while i > 0 and j > 0:
        if text1[i-1] == text2[j-1]:
            lcs.append(text1[i-1])
            i -= 1
            j -= 1
        elif dp[i-1][j] > dp[i][j-1]:
            i -= 1
        else:
            j -= 1

    return ''.join(reversed(lcs))

# 코드 비교 (diff 알고리즘)
code1 = "ABCDEFG"
code2 = "ABDXEFG"
common = lcs_string(code1, code2)
print(f"공통 부분: {common}")  # ABDEFG
```

```python
def min_coins(coins, amount):
    """최소 동전 개수로 거스름돈 만들기"""
    dp = [float('inf')] * (amount + 1)
    dp[0] = 0

    for i in range(1, amount + 1):
        for coin in coins:
            if coin <= i:
                dp[i] = min(dp[i], dp[i - coin] + 1)

    return dp[amount] if dp[amount] != float('inf') else -1

def coin_change_combinations(coins, amount):
    """거스름돈을 만드는 모든 경우의 수"""
    dp = [0] * (amount + 1)
    dp[0] = 1

    for coin in coins:
        for i in range(coin, amount + 1):
            dp[i] += dp[i - coin]

    return dp[amount]

# 사용
```

```python
class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = {}
        self.order = []  # 사용 순서 (가장 최근이 뒤)

    def get(self, key):
        if key not in self.cache:
            return -1

        # 최근 사용으로 갱신
        self.order.remove(key)
        self.order.append(key)

        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.order.remove(key)
        elif len(self.cache) >= self.capacity:
            # 가장 오래된 항목 제거
            oldest = self.order.pop(0)
            del self.cache[oldest]

        self.cache[key] = value
        self.order.append(key)
```

#### 실행 취소/다시 실행 (Undo/Redo)
```python
class UndoRedoManager:
    class Node:
        def __init__(self, action, data):
            self.action = action
            self.data = data
            self.prev = None
            self.next = None

    def __init__(self):
        self.current = None
        self.head = None

    def do_action(self, action, data):
        """새 작업 수행"""
        new_node = self.Node(action, data)

        if not self.head:
            self.head = new_node
            self.current = new_node
        else:
            # 현재 위치 이후의 작업들 제거 (새 브랜치)
            self.current.next = new_node
            new_node.prev = self.current
            self.current = new_node

    def undo(self):
        """실행 취소"""
        if not self.current or self.current == self.head:
            return None

        action = self.current.action
        data = self.current.data
        self.current = self.current.prev

        return (action, data)

    def redo(self):
        """다시 실행"""
        if not self.current or not self.current.next:
            return None

        self.current = self.current.next
        return (self.current.action, self.current.data)

# 사용 (텍스트 편집기)
editor = UndoRedoManager()
editor.do_action("INSERT", "Hello")
editor.do_action("INSERT", " World")
editor.do_action("DELETE", " World")

print(editor.undo())  # ("DELETE", " World")
print(editor.undo())  # ("INSERT", " World")
print(editor.redo())  # ("INSERT", " World")
```

```python
    """괄호 짝이 맞는지 검증"""
    stack = []
    pairs = {'(': ')', '[': ']', '{': '}'}

    for char in s:
        if char in pairs:  # 여는 괄호
            stack.append(char)
        elif char in pairs.values():  # 닫는 괄호
            if not stack:
                return False
            if pairs[stack.pop()] != char:
                return False

    return len(stack) == 0
```

```python
class BrowserHistory:
    def __init__(self):
        self.back_stack = []
        self.forward_stack = []
        self.current = None

    def visit(self, url):
        """새 페이지 방문"""
        if self.current:
            self.back_stack.append(self.current)
        self.current = url
        self.forward_stack = []  # 새 페이지 방문 시 forward 기록 삭제

    def back(self):
        """뒤로 가기"""
        if not self.back_stack:
            return self.current

        self.forward_stack.append(self.current)
        self.current = self.back_stack.pop()
        return self.current

    def forward(self):
        """앞으로 가기"""
        if not self.forward_stack:
            return self.current

        self.back_stack.append(self.current)
        self.current = self.forward_stack.pop()
        return self.current

# 사용
```

```python
        self.queue = deque()
        self.num_workers = num_workers

        """작업 추가 (우선순위 있음)"""
```

```python
class MedianFinder:
    def __init__(self):
        self.small = []  # 최대 힙 (음수로 저장)
        self.large = []  # 최소 힙

    def add_num(self, num):
        # 작은 쪽 힙에 추가 (최대 힙)
        heapq.heappush(self.small, -num)

        # 균형 맞추기
        if (self.small and self.large and
            -self.small[0] > self.large[0]):
            val = -heapq.heappop(self.small)
            heapq.heappush(self.large, val)

        # 크기 균형
        if len(self.small) > len(self.large) + 1:
            val = -heapq.heappop(self.small)
            heapq.heappush(self.large, val)
        if len(self.large) > len(self.small):
            val = heapq.heappop(self.large)
            heapq.heappush(self.small, -val)

    def find_median(self):
        if len(self.small) > len(self.large):
            return float(-self.small[0])

        return (-self.small[0] + self.large[0]) / 2.0

# 사용
```

```python
def merge_k_sorted_lists(lists):
    """K개의 정렬된 리스트를 하나로 병합"""
    import heapq

    heap = []
    result = []

    # 각 리스트의 첫 원소를 힙에 추가
    for i, lst in enumerate(lists):
        if lst:
            heapq.heappush(heap, (lst[0], i, 0))

    while heap:
        val, list_idx, elem_idx = heapq.heappop(heap)
        result.append(val)

        # 다음 원소 추가
        if elem_idx + 1 < len(lists[list_idx]):
            next_val = lists[list_idx][elem_idx + 1]
            heapq.heappush(heap, (next_val, list_idx, elem_idx + 1))

    return result

# 사용

    [1, 4, 5],
    [1, 3, 4],
    [2, 6]
]
```

```python
class AutoComplete:
    class TrieNode:
        def __init__(self):
            self.children = {}
            self.is_end = False
            self.frequency = 0  # 검색 빈도

    def __init__(self):
        self.root = self.TrieNode()

    def insert(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = self.TrieNode()
            node = node.children[char]

        node.is_end = True
        node.frequency += 1  # 빈도 증가

    def autocomplete(self, prefix, limit=5):
        """접두사로 시작하는 단어 추천 (빈도순)"""
        node = self.root

        # 접두사 위치로 이동
        for char in prefix:
            if char not in node.children:
                return []
            node = node.children[char]

        # 모든 단어 수집
        words = []
        self._collect_words(node, prefix, words)

        # 빈도순 정렬
        words.sort(key=lambda x: x[1], reverse=True)

        return [word for word, _ in words[:limit]]

    def _collect_words(self, node, prefix, words):
        if node.is_end:
            words.append((prefix, node.frequency))

        for char, child_node in node.children.items():
            self._collect_words(child_node, prefix + char, words)
```

```python
# N-Queens 문제
def solve_n_queens(n):
    def is_safe(board, row, col):
        # 같은 열 체크
        for i in range(row):
            if board[i] == col:
                return False

        # 대각선 체크
        for i in range(row):
            if abs(board[i] - col) == abs(i - row):
                return False

        return True

    def backtrack(row):
        if row == n:
            solutions.append(board[:])
            return

        for col in range(n):
            if is_safe(board, row, col):
                board[row] = col
                backtrack(row + 1)
                board[row] = -1  # 백트랙

    solutions = []
    board = [-1] * n
    backtrack(0)
    return solutions

# 4-Queens
```

```python
def solve_sudoku(board):
    """스도쿠 퍼즐 해결"""
    def is_valid(board, row, col, num):
        # 행 체크
        if num in board[row]:
            return False

        # 열 체크
        if num in [board[i][col] for i in range(9)]:
            return False

        # 3x3 박스 체크
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if board[i][j] == num:
                    return False

        return True

    def solve():
        for i in range(9):
            for j in range(9):
                if board[i][j] == 0:
                    for num in range(1, 10):
                        if is_valid(board, i, j, num):
                            board[i][j] = num

                            if solve():
                                return True

                            board[i][j] = 0  # 백트랙

                    return False
        return True

    solve()
    return board
```

```python
def generate_combinations(arr, k):
    """배열에서 k개를 선택하는 모든 조합"""
    result = []

    def backtrack(start, current):
        if len(current) == k:
            result.append(current[:])
            return

        for i in range(start, len(arr)):
            current.append(arr[i])
            backtrack(i + 1, current)
            current.pop()  # 백트랙

    backtrack(0, [])
    return result

# 사용
arr = [1, 2, 3, 4]
combinations = generate_combinations(arr, 2)
print(combinations)  # [[1,2], [1,3], [1,4], [2,3], [2,4], [3,4]]
```

#### 부분집합 합 문제
```python
def subset_sum(nums, target):
    """목표 합을 만드는 모든 부분집합 찾기"""
    result = []

    def backtrack(start, current, current_sum):
        if current_sum == target:
            result.append(current[:])
            return

        if current_sum > target:
            return

        for i in range(start, len(nums)):
            current.append(nums[i])
            backtrack(i + 1, current, current_sum + nums[i])
            current.pop()

    backtrack(0, [], 0)
    return result

# 사용
nums = [2, 3, 6, 7]
target = 7
subsets = subset_sum(nums, target)
print(f"합이 {target}인 부분집합:", subsets)
# [[7], [2, 2, 3]]
```

### 🎯 백트래킹 패턴

1. **선택 (Choose)**: 후보 선택
2. **탐색 (Explore)**: 재귀적으로 탐색
3. **철회 (Unchoose)**: 선택 취소 (백트랙)

**적용 사례:**
- 퍼즐 해결 (스도쿠, N-Queens)
- 조합/순열 생성
- 부분집합 문제
- 경로 찾기
