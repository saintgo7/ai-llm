### 🎯 실제 프로젝트 적용 방법

1. **기존 API에 인증 추가**
   ```python
   from functools import wraps
   import jwt

   def token_required(f):
       @wraps(f)
       def decorated(*args, **kwargs):
           token = request.headers.get('Authorization')
           if not token:
               return jsonify({'message': '토큰이 없습니다'}), 401

           try:
               token = token.split()[1]  # "Bearer TOKEN"
               data = jwt.decode(token, app.config['SECRET_KEY'],
                               algorithms=["HS256"])
           except:
               return jsonify({'message': '유효하지 않은 토큰'}), 401

           return f(*args, **kwargs)
       return decorated

   @app.route('/api/protected')
   @token_required
   def protected():
       return jsonify({'message': '인증 성공'})
   ```

2. **React 프론트엔드와 연동**
   ```javascript
   // 로그인
   const login = async (username, password) => {
     const response = await fetch('/api/login', {
       method: 'POST',
       headers: { 'Content-Type': 'application/json' },
       body: JSON.stringify({ username, password })
     });
     const data = await response.json();
     localStorage.setItem('token', data.token);
   };

   // API 호출
   const fetchData = async () => {
     const token = localStorage.getItem('token');
     const response = await fetch('/api/data', {
       headers: { 'Authorization': `Bearer ${token}` }
     });
     return response.json();
   };
   ```

3. **토큰 갱신 구현**
   ```python
   @app.route('/api/refresh', methods=['POST'])
   def refresh():
       token = request.headers.get('Authorization').split()[1]
       try:
           data = jwt.decode(token, app.config['SECRET_KEY'],
                           algorithms=["HS256"])
           new_token = jwt.encode(
               {'username': data['username'],
                'exp': datetime.utcnow() + timedelta(hours=1)},
               app.config['SECRET_KEY']
           )
           return jsonify({'token': new_token})
       except:
           return jsonify({'message': '토큰 갱신 실패'}), 401
   ```

```python
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

def send_email(to_email, subject, body):
    sender_email = "your-email@gmail.com"
    password = "your-app-password"  # Gmail 앱 비밀번호

    msg = MIMEMultipart()
    msg['From'] = sender_email
    msg['To'] = to_email
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    with smtplib.SMTP('smtp.gmail.com', 587) as server:
        server.starttls()
        server.login(sender_email, password)
        server.send_message(msg)

# 사용 예제
```

```python
from email.mime.base import MIMEBase
from email import encoders

def send_email_with_attachment(to_email, subject, body, filename):
    msg = MIMEMultipart()
    msg['From'] = sender_email
    msg['To'] = to_email
    msg['Subject'] = subject

    msg.attach(MIMEText(body, 'plain'))

    # 파일 첨부
    with open(filename, 'rb') as attachment:
        part = MIMEBase('application', 'octet-stream')
        part.set_payload(attachment.read())

    encoders.encode_base64(part)
    part.add_header(
        'Content-Disposition',
        f'attachment; filename= {filename}'
    )

    msg.attach(part)

    with smtplib.SMTP('smtp.gmail.com', 587) as server:
        server.starttls()
        server.login(sender_email, password)
        server.send_message(msg)

# 사용
send_email_with_attachment(
    'client@example.com',
    '월간 리포트',
    '첨부된 리포트를 확인해 주세요.',
```

### 💻 실전 사용 예제

#### 파일 암호화
```python
from cryptography.fernet import Fernet

# 키 생성 (한 번만 실행, 안전하게 보관)
key = Fernet.generate_key()
with open('secret.key', 'wb') as key_file:
    key_file.write(key)

# 암호화
def encrypt_file(filename, key):
    fernet = Fernet(key)

    with open(filename, 'rb') as file:
        original = file.read()

    encrypted = fernet.encrypt(original)

    with open(filename + '.encrypted', 'wb') as encrypted_file:
        encrypted_file.write(encrypted)

    print(f"파일 암호화 완료: {filename}.encrypted")

# 사용
with open('secret.key', 'rb') as key_file:
    key = key_file.read()

encrypt_file('confidential.pdf', key)
```

#### 파일 복호화
```python
def decrypt_file(filename, key):
    fernet = Fernet(key)

    with open(filename, 'rb') as encrypted_file:
        encrypted = encrypted_file.read()

    decrypted = fernet.decrypt(encrypted)

    output_filename = filename.replace('.encrypted', '.decrypted')
    with open(output_filename, 'wb') as decrypted_file:
        decrypted_file.write(decrypted)

    print(f"파일 복호화 완료: {output_filename}")

# 사용
decrypt_file('confidential.pdf.encrypted', key)
```

#### 텍스트 암호화
```python
def encrypt_text(text, key):
    fernet = Fernet(key)
    encrypted_text = fernet.encrypt(text.encode())
    return encrypted_text

def decrypt_text(encrypted_text, key):
    fernet = Fernet(key)
    decrypted_text = fernet.decrypt(encrypted_text).decode()
    return decrypted_text

# 사용
secret_message = "이것은 비밀 메시지입니다"
encrypted = encrypt_text(secret_message, key)
print(f"암호화: {encrypted}")

decrypted = decrypt_text(encrypted, key)
print(f"복호화: {decrypted}")
```

```python
    def __init__(self):
        self.patterns = {
            r'(안녕|hello|hi)': ['안녕하세요! 무엇을 도와드릴까요?'],
            r'(주문|order).*확인': ['주문번호를 알려주시면 확인해 드리겠습니다.'],
```

```python
import secrets
import string

def generate_strong_password(length=16):
    # 문자 집합
    alphabet = string.ascii_letters + string.digits + string.punctuation

    # 최소 조건: 대문자 1개, 소문자 1개, 숫자 1개, 특수문자 1개
    password = [
        secrets.choice(string.ascii_uppercase),
        secrets.choice(string.ascii_lowercase),
        secrets.choice(string.digits),
        secrets.choice(string.punctuation)
    ]

    # 나머지 문자 랜덤 생성
    for _ in range(length - 4):
        password.append(secrets.choice(alphabet))

    # 섞기
    secrets.SystemRandom().shuffle(password)

    return ''.join(password)
```

   ```python
       def __init__(self, master_password, totp_secret=None):
           super().__init__(master_password)
           self.totp_secret = totp_secret or pyotp.random_base32()
           self.totp = pyotp.TOTP(self.totp_secret)

       def verify_2fa(self, code):
           return self.totp.verify(code)

       def get_password(self, service, totp_code):
           if not self.verify_2fa(totp_code):
               raise Exception("2단계 인증 실패")

           return super().get_password(service)
   ```

3. **자동 잠금**
   ```python
   import time

   class AutoLockPasswordManager:
       def __init__(self, master_password, lock_timeout=300):
           self.pm = PasswordManager(master_password)
           self.lock_timeout = lock_timeout  # 5분
           self.last_access = time.time()
           self.is_locked = False

       def access(self):
           if self.is_locked or (time.time() - self.last_access) > self.lock_timeout:
               raise Exception("세션이 만료되었습니다. 다시 로그인하세요.")

           self.last_access = time.time()

       def get_password(self, service):
           self.access()
           return self.pm.get_password(service)
   ```
