```python
    """
    Minimax 알고리즘
    - 모든 가능한 수를 시뮬레이션
    - 최선의 수를 선택
    """
    # 게임 종료 상태 체크
    winner = check_winner(board)
    if winner == 'O':  # AI 승리
        return 10 - depth
    elif winner == 'X':  # 플레이어 승리
        return depth - 10
    elif is_board_full(board):  # 무승부
        return 0

    if is_maximizing:  # AI 차례
        best_score = -float('inf')
        for move in get_available_moves(board):
            board[move] = 'O'
            score = minimax(board, depth + 1, False)
            board[move] = ' '
            best_score = max(score, best_score)
        return best_score
    else:  # 플레이어 차례
        best_score = float('inf')
        for move in get_available_moves(board):
            board[move] = 'X'
            score = minimax(board, depth + 1, True)
            board[move] = ' '
            best_score = min(score, best_score)
        return best_score
```

```python
class SnakeGame:
    def __init__(self, width=20, height=10):
        self.width = width
        self.height = height
        self.snake = [(width//2, height//2)]  # 뱀의 몸
        self.direction = (1, 0)  # 오른쪽
        self.food = self.spawn_food()
        self.score = 0

    def move(self):
        """뱀 이동"""
        head_x, head_y = self.snake[0]
        dir_x, dir_y = self.direction

        # 새 머리 위치
        new_head = (head_x + dir_x, head_y + dir_y)

        # 벽 충돌 체크
        if (new_head[0] < 0 or new_head[0] >= self.width or
            new_head[1] < 0 or new_head[1] >= self.height):
            return False  # 게임 오버

        # 자기 몸 충돌 체크
        if new_head in self.snake:
            return False  # 게임 오버

        # 먹이 먹었는지 체크
        if new_head == self.food:
            self.snake.insert(0, new_head)
            self.food = self.spawn_food()
            self.score += 10
        else:
            self.snake.insert(0, new_head)
            self.snake.pop()  # 꼬리 제거

        return True

    def spawn_food(self):
        """먹이 생성"""
        while True:
            food = (random.randint(0, self.width-1),
                   random.randint(0, self.height-1))
            if food not in self.snake:
                return food
```

```python
class Hangman:
    WORDS = ['python', 'programming', 'computer', 'algorithm', 'database']

    def __init__(self):
        self.word = random.choice(self.WORDS)
        self.guessed_letters = set()
        self.wrong_guesses = 0
        self.max_wrong = 6

    def guess(self, letter):
        """글자 맞추기"""
        letter = letter.lower()

        if letter in self.guessed_letters:
            return "이미 입력한 글자입니다"

        self.guessed_letters.add(letter)

        if letter in self.word:
            if self.is_won():
                return "WIN"
            return "CORRECT"
        else:
            self.wrong_guesses += 1
            if self.wrong_guesses >= self.max_wrong:
                return "LOSE"
            return "WRONG"

    def get_display_word(self):
        """현재 진행 상황 표시"""
        return ' '.join(
            letter if letter in self.guessed_letters else '_'
            for letter in self.word
        )

    def is_won(self):
        """승리 확인"""
        return all(letter in self.guessed_letters for letter in self.word)
```

### 🎮 조작법

### 💻 핵심 로직

```python
class Game2048:
    def __init__(self, size=4):
        self.size = size
        self.board = [[0] * size for _ in range(size)]
        self.score = 0
        self.add_new_tile()
        self.add_new_tile()

    def move_left(self):
        """왼쪽으로 이동"""
        moved = False
        for i in range(self.size):
            # 0이 아닌 숫자만 모으기
            row = [num for num in self.board[i] if num != 0]

            # 같은 숫자 합치기
            j = 0
            while j < len(row) - 1:
                if row[j] == row[j + 1]:
                    row[j] *= 2
                    self.score += row[j]
                    row.pop(j + 1)
                    moved = True
                j += 1

            # 0으로 패딩
            row += [0] * (self.size - len(row))

            if row != self.board[i]:
                moved = True

            self.board[i] = row

        if moved:
            self.add_new_tile()

        return moved

    def add_new_tile(self):
        """새 타일 추가 (2 또는 4)"""
        empty_cells = [
            (i, j)
            for i in range(self.size)
            for j in range(self.size)
            if self.board[i][j] == 0
        ]

        if empty_cells:
            i, j = random.choice(empty_cells)
            self.board[i][j] = 2 if random.random() < 0.9 else 4

    def is_game_over(self):
        """게임 오버 확인"""
        # 빈 칸이 있으면 게임 계속
        for row in self.board:
            if 0 in row:
                return False

        # 합칠 수 있는 타일이 있는지 확인
        for i in range(self.size):
            for j in range(self.size):
                current = self.board[i][j]

                # 오른쪽 확인
                if j < self.size - 1 and current == self.board[i][j + 1]:
                    return False

                # 아래 확인
                if i < self.size - 1 and current == self.board[i + 1][j]:
                    return False

        return True
```

```python
class AI2048:
    def get_best_move(self, game):
        """최선의 수 찾기"""
        best_score = -1
        best_move = None

        for move in ['W', 'A', 'S', 'D']:
            # 시뮬레이션
            test_game = copy.deepcopy(game)

            if test_game.move(move):
                score = self.evaluate_board(test_game.board)

                if score > best_score:
                    best_score = score
                    best_move = move

        return best_move

    def evaluate_board(self, board):
        """보드 평가 (휴리스틱)"""
        score = 0

        # 빈 칸 개수 (많을수록 좋음)
        empty_cells = sum(row.count(0) for row in board)
        score += empty_cells * 10

        # 큰 숫자가 모서리에 있으면 좋음
        score += board[0][0] * 5

        # 숫자가 감소하는 순서로 정렬되어 있으면 좋음
        for i in range(len(board)):
            for j in range(len(board[i]) - 1):
                if board[i][j] >= board[i][j + 1]:
                    score += 1

        return score
```

### 💻 해결 알고리즘

```python
class SudokuSolver:
    def __init__(self, board):
        self.board = board

    def is_valid(self, row, col, num):
        """숫자가 유효한지 확인"""
        # 행 체크
        if num in self.board[row]:
            return False

        # 열 체크
        if num in [self.board[i][col] for i in range(9)]:
            return False

        # 3x3 박스 체크
        box_row, box_col = 3 * (row // 3), 3 * (col // 3)
        for i in range(box_row, box_row + 3):
            for j in range(box_col, box_col + 3):
                if self.board[i][j] == num:
                    return False

        return True

    def solve(self):
        """백트래킹으로 해결"""
        # 빈 칸 찾기
        for i in range(9):
            for j in range(9):
                if self.board[i][j] == 0:
                    # 1-9 시도
                    for num in range(1, 10):
                        if self.is_valid(i, j, num):
                            self.board[i][j] = num

                            # 재귀적으로 다음 칸 해결
                            if self.solve():
                                return True

                            # 백트랙
                            self.board[i][j] = 0

                    # 어떤 숫자도 안 됨
                    return False

        # 모든 칸이 채워짐
        return True
```

### 🎲 스도쿠 생성기

```python
class SudokuGenerator:
    def generate(self, difficulty='medium'):
        """난이도별 스도쿠 생성"""
        # 빈 보드 생성
        board = [[0] * 9 for _ in range(9)]

        # 대각선 3x3 박스 채우기
        for i in range(0, 9, 3):
            self.fill_box(board, i, i)

        # 나머지 해결
        solver = SudokuSolver(board)
        solver.solve()

        # 난이도에 따라 숫자 제거
        remove_count = {
            'easy': 30,
            'medium': 40,
            'hard': 50,
            'expert': 60
        }[difficulty]

        self.remove_numbers(board, remove_count)

        return board

    def fill_box(self, board, row, col):
        """3x3 박스를 랜덤하게 채우기"""
        numbers = list(range(1, 10))
        random.shuffle(numbers)

        for i in range(3):
            for j in range(3):
                board[row + i][col + j] = numbers.pop()

    def remove_numbers(self, board, count):
        """숫자 제거"""
        cells = [(i, j) for i in range(9) for j in range(9)]
        random.shuffle(cells)

        for i, j in cells[:count]:
            board[i][j] = 0
```

## 6. 체스 엔진

**파일**: `46_chess_engine.py`

### 📖 개요
기본적인 체스 규칙과 이동 검증을 구현한 체스 엔진입니다.

### 💻 핵심 구현

```python
class ChessBoard:
    def __init__(self):
        self.board = self.initialize_board()
        self.current_turn = 'white'

    def initialize_board(self):
        """초기 보드 설정"""
        board = [[None] * 8 for _ in range(8)]

        # 폰
        for i in range(8):
            board[1][i] = Piece('pawn', 'black')
            board[6][i] = Piece('pawn', 'white')

        # 룩
        board[0][0] = board[0][7] = Piece('rook', 'black')
        board[7][0] = board[7][7] = Piece('rook', 'white')

        # 나이트
        board[0][1] = board[0][6] = Piece('knight', 'black')
        board[7][1] = board[7][6] = Piece('knight', 'white')

        # 비숍
        board[0][2] = board[0][5] = Piece('bishop', 'black')
        board[7][2] = board[7][5] = Piece('bishop', 'white')

        # 퀸
        board[0][3] = Piece('queen', 'black')
        board[7][3] = Piece('queen', 'white')

        # 킹
        board[0][4] = Piece('king', 'black')
        board[7][4] = Piece('king', 'white')

        return board

    def is_valid_move(self, from_pos, to_pos):
        """이동이 유효한지 확인"""
        from_row, from_col = from_pos
        to_row, to_col = to_pos

        piece = self.board[from_row][from_col]

        # 빈 칸
        if piece is None:
            return False

        # 차례 확인
        if piece.color != self.current_turn:
            return False

        # 같은 색 기물 공격 불가
        target = self.board[to_row][to_col]
        if target and target.color == piece.color:
            return False

        # 기물별 이동 규칙 체크
        return piece.can_move(from_pos, to_pos, self.board)

class Piece:
    def __init__(self, piece_type, color):
        self.type = piece_type
        self.color = color

    def can_move(self, from_pos, to_pos, board):
        """기물별 이동 가능 여부"""
        if self.type == 'pawn':
            return self.can_move_pawn(from_pos, to_pos, board)
        elif self.type == 'rook':
            return self.can_move_rook(from_pos, to_pos, board)
        elif self.type == 'knight':
            return self.can_move_knight(from_pos, to_pos)
        elif self.type == 'bishop':
            return self.can_move_bishop(from_pos, to_pos, board)
        elif self.type == 'queen':
            return self.can_move_queen(from_pos, to_pos, board)
        elif self.type == 'king':
            return self.can_move_king(from_pos, to_pos)

    def can_move_knight(self, from_pos, to_pos):
        """나이트 이동: L자 형태"""
        row_diff = abs(to_pos[0] - from_pos[0])
        col_diff = abs(to_pos[1] - from_pos[1])

        return (row_diff == 2 and col_diff == 1) or \
               (row_diff == 1 and col_diff == 2)
```

### 💻 DFS 미로 생성

```python
class MazeGenerator:
    def __init__(self, width, height):
        self.width = width
        self.height = height
        self.maze = [['#'] * width for _ in range(height)]

    def generate_dfs(self):
        """DFS로 미로 생성"""
        # 시작점
        start = (0, 0)
        self.maze[0][0] = ' '

        stack = [start]
        visited = {start}

        while stack:
            current = stack[-1]
            row, col = current

            # 방문하지 않은 이웃 찾기
            neighbors = []
            for dr, dc in [(0, 2), (2, 0), (0, -2), (-2, 0)]:
                new_row, new_col = row + dr, col + dc

                if (0 <= new_row < self.height and
                    0 <= new_col < self.width and
                    (new_row, new_col) not in visited):
                    neighbors.append((new_row, new_col))

            if neighbors:
                # 랜덤 이웃 선택
                next_cell = random.choice(neighbors)
                next_row, next_col = next_cell

                # 벽 제거
                wall_row = (row + next_row) // 2
                wall_col = (col + next_col) // 2
                self.maze[wall_row][wall_col] = ' '
                self.maze[next_row][next_col] = ' '

                visited.add(next_cell)
                stack.append(next_cell)
            else:
                stack.pop()

    def solve_bfs(self):
        """BFS로 미로 해결"""
        from collections import deque

        start = (0, 0)
        end = (self.height - 1, self.width - 1)

        queue = deque([(start, [start])])
        visited = {start}

        while queue:
            current, path = queue.popleft()

            if current == end:
                return path

            row, col = current

            for dr, dc in [(0, 1), (1, 0), (0, -1), (-1, 0)]:
                new_row, new_col = row + dr, col + dc

                if (0 <= new_row < self.height and
                    0 <= new_col < self.width and
                    self.maze[new_row][new_col] == ' ' and
                    (new_row, new_col) not in visited):

                    visited.add((new_row, new_col))
                    queue.append(((new_row, new_col), path + [(new_row, new_col)]))

        return None
```

## 8. 정규식 엔진

**파일**: `48_regex_engine.py`

### 📖 개요
간단한 정규식 패턴 매칭 엔진 구현입니다.

### 💻 핵심 개념

```python
class SimpleRegex:
    def match(self, pattern, text):
        """정규식 매칭"""
        if not pattern:
            return True

        # 첫 글자 매칭
        first_match = bool(text) and pattern[0] in {text[0], '.'}

        # * 처리
        if len(pattern) >= 2 and pattern[1] == '*':
            # 0개 매칭 또는 1개 이상 매칭
            return (self.match(pattern[2:], text) or
                   (first_match and self.match(pattern, text[1:])))

        # 일반 매칭
        return first_match and self.match(pattern[1:], text[1:])

# 사용
regex = SimpleRegex()
print(regex.match("a*b", "aaab"))  # True
print(regex.match("a.c", "abc"))   # True
```

```python
class MarkdownParser:
    def parse(self, markdown_text):
        """Markdown → HTML"""
        html = []

        for line in markdown_text.split('\n'):
            # 헤더
            if line.startswith('#'):
                level = len(line) - len(line.lstrip('#'))
                text = line.lstrip('#').strip()
                html.append(f'<h{level}>{text}</h{level}>')

            # 리스트
            elif line.strip().startswith('- '):
                text = line.strip()[2:]
                html.append(f'<li>{text}</li>')

            # 일반 텍스트
            elif line.strip():
                text = self.parse_inline(line.strip())
                html.append(f'<p>{text}</p>')

        return '\n'.join(html)

    def parse_inline(self, text):
        """인라인 요소 파싱"""
        # **굵게**
        text = re.sub(r'\*\*(.+?)\*\*', r'<strong>\1</strong>', text)

        # *기울임*
        text = re.sub(r'\*(.+?)\*', r'<em>\1</em>', text)

        # `코드`
        text = re.sub(r'`(.+?)`', r'<code>\1</code>', text)

        # [링크](url)
        text = re.sub(r'\[(.+?)\]\((.+?)\)', r'<a href="\2">\1</a>', text)

        return text
```

## 10. 컴파일러

**파일**: `50_compiler.py`

### 📖 개요
간단한 프로그래밍 언어의 렉서, 파서, 인터프리터 구현입니다.

### 💻 전체 구조

```python
# 1. 렉서 (Lexer) - 토큰화
class Lexer:
    def tokenize(self, code):
        """코드를 토큰으로 분리"""
        tokens = []
        # 숫자, 연산자, 키워드 등 인식
        return tokens

# 2. 파서 (Parser) - AST 생성
class Parser:
    def parse(self, tokens):
        """토큰을 AST로 변환"""
        ast = self.build_ast(tokens)
        return ast

# 3. 인터프리터 (Interpreter) - 실행
class Interpreter:
    def execute(self, ast):
        """AST를 실행"""
        result = self.evaluate(ast)
        return result
```

### 🎯 간단한 계산기 언어

```python
class SimpleCompiler:
    def compile_and_run(self, code):
        """
        지원 문법:
        - 숫자: 1, 2, 3.14
        - 연산: +, -, *, /
        - 변수: x = 10
        - 출력: print(x)
        """
        lexer = Lexer()
        tokens = lexer.tokenize(code)

        parser = Parser()
        ast = parser.parse(tokens)

        interpreter = Interpreter()
        result = interpreter.execute(ast)

        return result

# 사용
compiler = SimpleCompiler()

code = """
x = 10
y = 20
result = x + y
print(result)
"""

compiler.compile_and_run(code)
# 출력: 30
```
