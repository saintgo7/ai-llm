### 💻 실전 사용 예제

#### 로그인 자동화
```python
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
import time

class WebAutomation:
    def __init__(self, headless=True):
        options = webdriver.ChromeOptions()
        if headless:
            options.add_argument('--headless')
        options.add_argument('--no-sandbox')
        options.add_argument('--disable-dev-shm-usage')

        self.driver = webdriver.Chrome(options=options)
        self.wait = WebDriverWait(self.driver, 10)

    def login(self, url, username, password):
        """웹사이트 로그인"""
        try:
            self.driver.get(url)

            # 사용자명 입력
            username_field = self.wait.until(
                EC.presence_of_element_located((By.NAME, 'username'))
            )
            username_field.send_keys(username)

            # 비밀번호 입력
            password_field = self.driver.find_element(By.NAME, 'password')
            password_field.send_keys(password)

            # 로그인 버튼 클릭
            login_button = self.driver.find_element(By.CSS_SELECTOR, 'button[type="submit"]')
            login_button.click()

            # 로그인 완료 대기
            self.wait.until(EC.url_changes(url))

            print("✅ 로그인 성공")
            return True

        except Exception as e:
            print(f"❌ 로그인 실패: {e}")
            return False

    def screenshot(self, filename='screenshot.png'):
        """스크린샷 저장"""
        self.driver.save_screenshot(filename)
        print(f"스크린샷 저장: {filename}")

    def close(self):
        """브라우저 종료"""
        self.driver.quit()

# 사용
bot = WebAutomation(headless=False)
bot.login('https://example.com/login', 'myuser', 'mypass')
bot.screenshot('after_login.png')
bot.close()
```

```python
class WebScraper:
    def __init__(self):
        self.driver = webdriver.Chrome(ChromeDriverManager().install())

    def scrape_product_list(self, url):
        """상품 목록 스크래핑"""
        self.driver.get(url)

        # 페이지 스크롤 (동적 로딩)
        last_height = self.driver.execute_script("return document.body.scrollHeight")

        while True:
            # 스크롤 다운
            self.driver.execute_script("window.scrollTo(0, document.body.scrollHeight);")
            time.sleep(2)

            # 새 높이 확인
            new_height = self.driver.execute_script("return document.body.scrollHeight")
            if new_height == last_height:
                break
            last_height = new_height

        # 상품 정보 수집
        products = []
        items = self.driver.find_elements(By.CLASS_NAME, 'product-item')

        for item in items:
            try:
                name = item.find_element(By.CLASS_NAME, 'product-name').text
                price = item.find_element(By.CLASS_NAME, 'product-price').text

                products.append({
                    'name': name,
                    'price': price
                })
            except:
                continue

        return products

    def fill_form(self, form_data):
        """폼 자동 작성"""
        for field_name, value in form_data.items():
            try:
                field = self.driver.find_element(By.NAME, field_name)
                field.clear()
                field.send_keys(value)
            except:
                print(f"필드를 찾을 수 없음: {field_name}")

# 사용
scraper = WebScraper()
products = scraper.scrape_product_list('https://shop.example.com/products')

for product in products[:10]:
    print(f"{product['name']}: {product['price']}")
```

```python
from reportlab.lib.pagesizes import A4
from reportlab.pdfgen import canvas
from reportlab.lib.units import cm
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from datetime import datetime

class InvoiceGenerator:
    def __init__(self):
        # 한글 폰트 등록
        try:
            pdfmetrics.registerFont(TTFont('NanumGothic', 'NanumGothic.ttf'))
            self.font = 'NanumGothic'
        except:
            self.font = 'Helvetica'

    def create_invoice(self, filename, invoice_data):
        """송장 PDF 생성"""
        c = canvas.Canvas(filename, pagesize=A4)
        width, height = A4

        # 제목
        c.setFont(self.font, 24)
        c.drawString(50, height - 50, "송 장")

        # 날짜
        c.setFont(self.font, 10)
        c.drawString(50, height - 80, f"발행일: {datetime.now().strftime('%Y년 %m월 %d일')}")

        # 송장 번호
        c.drawString(50, height - 95, f"송장번호: {invoice_data['invoice_no']}")

        # 선 그리기
        c.line(50, height - 110, width - 50, height - 110)

        # 고객 정보
        c.setFont(self.font, 12)
        c.drawString(50, height - 140, "받는 사람:")
        c.setFont(self.font, 10)
        c.drawString(70, height - 160, f"이름: {invoice_data['customer_name']}")
        c.drawString(70, height - 175, f"주소: {invoice_data['customer_address']}")

        # 항목 테이블
        y = height - 220
        c.setFont(self.font, 10)
        c.drawString(50, y, "항목")
        c.drawString(250, y, "수량")
        c.drawString(350, y, "단가")
        c.drawString(450, y, "금액")

        y -= 20
        c.line(50, y, width - 50, y)

        # 항목 리스트
        y -= 20
        total = 0
        for item in invoice_data['items']:
            c.drawString(50, y, item['name'])
            c.drawString(250, y, str(item['quantity']))
            c.drawString(350, y, f"{item['price']:,}원")
            amount = item['quantity'] * item['price']
            c.drawString(450, y, f"{amount:,}원")
            total += amount
            y -= 20

        # 합계
        y -= 10
        c.line(50, y, width - 50, y)
        y -= 20
        c.setFont(self.font, 12)
        c.drawString(350, y, "합계:")
        c.drawString(450, y, f"{total:,}원")

        # 하단 정보
        c.setFont(self.font, 8)
        c.drawString(50, 50, "회사명: (주)예시컴퍼니 | 전화: 02-1234-5678 | 이메일: info@example.com")

        c.save()
        print(f"송장 생성 완료: {filename}")
```

#### 리포트 생성
```python
from reportlab.lib.pagesizes import A4
from reportlab.platypus import SimpleDocTemplate, Table, TableStyle, Paragraph, Spacer
from reportlab.lib.styles import getSampleStyleSheet
from reportlab.lib import colors

class ReportGenerator:
    def __init__(self):
        self.styles = getSampleStyleSheet()

    def create_sales_report(self, filename, data):
        """매출 리포트 생성"""
        doc = SimpleDocTemplate(filename, pagesize=A4)
        story = []

        # 제목
        title = Paragraph("월간 매출 리포트", self.styles['Title'])
        story.append(title)
        story.append(Spacer(1, 20))

        # 요약
        summary = Paragraph(f"""
        <b>기간:</b> 2024년 11월<br/>
        <b>총 매출:</b> {data['total_sales']:,}원<br/>
        <b>거래 건수:</b> {data['transaction_count']:,}건<br/>
        <b>평균 객단가:</b> {data['average_order']:,}원
        """, self.styles['Normal'])
        story.append(summary)
        story.append(Spacer(1, 20))

        # 테이블 데이터
        table_data = [['날짜', '매출', '거래 건수']]
        for row in data['daily_sales']:
            table_data.append([
                row['date'],
                f"{row['sales']:,}원",
                f"{row['count']:,}건"
            ])

        # 테이블 생성
        table = Table(table_data)
        table.setStyle(TableStyle([
            ('BACKGROUND', (0, 0), (-1, 0), colors.grey),
            ('TEXTCOLOR', (0, 0), (-1, 0), colors.whitesmoke),
            ('ALIGN', (0, 0), (-1, -1), 'CENTER'),
            ('FONTNAME', (0, 0), (-1, 0), 'Helvetica-Bold'),
            ('FONTSIZE', (0, 0), (-1, 0), 12),
            ('BOTTOMPADDING', (0, 0), (-1, 0), 12),
            ('BACKGROUND', (0, 1), (-1, -1), colors.beige),
            ('GRID', (0, 0), (-1, -1), 1, colors.black)
        ]))

        story.append(table)

        # PDF 생성
        doc.build(story)
        print(f"리포트 생성 완료: {filename}")

# 사용

report_data = {
    'total_sales': 15500000,
    'transaction_count': 320,
    'average_order': 48437,
    'daily_sales': [
        {'date': '2024-11-01', 'sales': 500000, 'count': 12},
        {'date': '2024-11-02', 'sales': 750000, 'count': 18},
        # ... 더 많은 데이터
    ]
```

```python
import click
from pathlib import Path
import shutil
from rich.console import Console
from rich.table import Table

console = Console()

@click.group()
def cli():
    """파일 관리 도구"""
    pass

@cli.command()
@click.argument('directory')
@click.option('--recursive', '-r', is_flag=True, help='하위 디렉토리 포함')
def list(directory, recursive):
    """파일 목록 표시"""
    path = Path(directory)

    if not path.exists():
        console.print(f"[red]디렉토리가 존재하지 않음: {directory}[/red]")
        return

    table = Table(title=f"파일 목록: {directory}")
    table.add_column("이름", style="cyan")
    table.add_column("크기", style="magenta")
    table.add_column("수정일", style="green")

    pattern = '**/*' if recursive else '*'
    for item in path.glob(pattern):
        if item.is_file():
            size = item.stat().st_size
            mtime = item.stat().st_mtime
            table.add_row(
                item.name,
                f"{size:,} bytes",
                time.strftime('%Y-%m-%d %H:%M', time.localtime(mtime))
            )

    console.print(table)

@cli.command()
@click.argument('source')
@click.argument('destination')
def copy(source, destination):
    """파일 복사"""
    try:
        shutil.copy2(source, destination)
        console.print(f"[green]복사 완료: {source} → {destination}[/green]")
    except Exception as e:
        console.print(f"[red]복사 실패: {e}[/red]")

@cli.command()
@click.argument('pattern')
@click.option('--directory', '-d', default='.', help='검색 디렉토리')
def search(pattern, directory):
    """파일 검색"""
    path = Path(directory)
    found = list(path.rglob(f'*{pattern}*'))

    if not found:
        console.print(f"[yellow]'{pattern}'과 일치하는 파일이 없습니다[/yellow]")
        return

    console.print(f"[green]{len(found)}개 파일 발견:[/green]")
    for file in found:
        console.print(f"  📄 {file}")

if __name__ == '__main__':
    cli()
```

#### 데이터베이스 관리 CLI
```python
import click
import sqlite3
from rich.console import Console
from rich.table import Table

console = Console()

@click.group()
@click.option('--db', default='app.db', help='데이터베이스 파일')
@click.pass_context
def db_cli(ctx, db):
    """데이터베이스 관리 도구"""
    ctx.ensure_object(dict)
    ctx.obj['db'] = db

@db_cli.command()
@click.pass_context
def init(ctx):
    """데이터베이스 초기화"""
    conn = sqlite3.connect(ctx.obj['db'])
    cursor = conn.cursor()

    cursor.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY,
            username TEXT UNIQUE,
            email TEXT
        )
    ''')

    conn.commit()
    conn.close()
    console.print("[green]데이터베이스 초기화 완료[/green]")

@db_cli.command()
@click.argument('username')
@click.argument('email')
@click.pass_context
def add_user(ctx, username, email):
    """사용자 추가"""
    conn = sqlite3.connect(ctx.obj['db'])
    cursor = conn.cursor()

    try:
        cursor.execute(
            "INSERT INTO users (username, email) VALUES (?, ?)",
            (username, email)
        )
        conn.commit()
        console.print(f"[green]사용자 추가됨: {username}[/green]")
    except sqlite3.IntegrityError:
        console.print(f"[red]사용자명이 이미 존재합니다: {username}[/red]")
    finally:
        conn.close()

@db_cli.command()
@click.pass_context
def list_users(ctx):
    """사용자 목록"""
    conn = sqlite3.connect(ctx.obj['db'])
    cursor = conn.cursor()

    cursor.execute("SELECT * FROM users")
    users = cursor.fetchall()

    if not users:
        console.print("[yellow]사용자가 없습니다[/yellow]")
        return

    table = Table(title="사용자 목록")
    table.add_column("ID", style="cyan")
    table.add_column("사용자명", style="magenta")
    table.add_column("이메일", style="green")

    for user in users:
        table.add_row(str(user[0]), user[1], user[2])

    console.print(table)
    conn.close()

if __name__ == '__main__':
    db_cli()
```

## 4. 설정 관리자

**파일**: `34_config_manager.py`

### 📖 개요
다양한 형식의 설정 파일을 관리하는 도구입니다.

### 🚀 빠른 시작

```python
import json
import yaml
import configparser

class ConfigManager:
    def __init__(self, config_file):
        self.config_file = config_file
        self.config = {}

    def load(self):
        """설정 파일 로드"""
        if self.config_file.endswith('.json'):
            with open(self.config_file, 'r') as f:
                self.config = json.load(f)
        elif self.config_file.endswith('.yaml'):
            with open(self.config_file, 'r') as f:
                self.config = yaml.safe_load(f)
        elif self.config_file.endswith('.ini'):
            config = configparser.ConfigParser()
            config.read(self.config_file)
            self.config = {s: dict(config.items(s)) for s in config.sections()}

        return self.config

    def get(self, key, default=None):
        """설정 값 가져오기"""
        return self.config.get(key, default)

    def set(self, key, value):
        """설정 값 설정"""
        self.config[key] = value

    def save(self):
        """설정 파일 저장"""
        if self.config_file.endswith('.json'):
            with open(self.config_file, 'w') as f:
                json.dump(self.config, f, indent=2)

# 사용
config = ConfigManager('app.json')
config.load()
db_host = config.get('database.host', 'localhost')
```
