#### 사용자 관리 시스템
```python
class UserDatabase:
    def __init__(self, db_name='users.db'):
        self.db_name = db_name
        self.init_database()

    def init_database(self):
        """데이터베이스 초기화"""
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS users (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    username TEXT NOT NULL UNIQUE,
                    email TEXT NOT NULL,
                    password_hash TEXT NOT NULL,
                    role TEXT DEFAULT 'user',
                    is_active BOOLEAN DEFAULT 1,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    last_login TIMESTAMP
                )
            ''')

            cursor.execute('''
                CREATE INDEX IF NOT EXISTS idx_username
                ON users(username)
            ''')

            cursor.execute('''
                CREATE INDEX IF NOT EXISTS idx_email
                ON users(email)
            ''')

    def create_user(self, username, email, password):
        """새 사용자 생성"""
        import hashlib

        password_hash = hashlib.sha256(password.encode()).hexdigest()

        try:
            with sqlite3.connect(self.db_name) as conn:
                cursor = conn.cursor()
                cursor.execute(
                    "INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)",
                    (username, email, password_hash)
                )
                return cursor.lastrowid
        except sqlite3.IntegrityError:
            raise ValueError(f"사용자명 '{username}'이 이미 존재합니다")

    def get_user(self, username):
        """사용자 조회"""
        with sqlite3.connect(self.db_name) as conn:
            conn.row_factory = sqlite3.Row
            cursor = conn.cursor()
            cursor.execute(
                "SELECT * FROM users WHERE username = ?",
                (username,)
            )
            row = cursor.fetchone()
            return dict(row) if row else None

    def update_last_login(self, username):
        """마지막 로그인 시간 업데이트"""
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute(
                "UPDATE users SET last_login = CURRENT_TIMESTAMP WHERE username = ?",
                (username,)
            )

    def get_all_users(self, active_only=True):
        """모든 사용자 조회"""
        query = "SELECT id, username, email, role, created_at FROM users"
        if active_only:
            query += " WHERE is_active = 1"

        with sqlite3.connect(self.db_name) as conn:
            conn.row_factory = sqlite3.Row
            cursor = conn.cursor()
            cursor.execute(query)
            return [dict(row) for row in cursor.fetchall()]

    def delete_user(self, username):
        """사용자 삭제 (소프트 삭제)"""
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute(
                "UPDATE users SET is_active = 0 WHERE username = ?",
                (username,)
            )
            return cursor.rowcount > 0

# 사용
db = UserDatabase()

# 사용자 생성
user_id = db.create_user("kim", "kim@example.com", "password123")
print(f"생성된 사용자 ID: {user_id}")

# 사용자 조회
user = db.get_user("kim")
print(f"사용자 정보: {user}")

# 전체 사용자 목록
users = db.get_all_users()
for u in users:
    print(f"- {u['username']} ({u['email']})")
```

```python
class AccountDatabase:
    def __init__(self, db_name='accounts.db'):
        self.db_name = db_name
        self.init_database()

    def init_database(self):
        with sqlite3.connect(self.db_name) as conn:
            cursor = conn.cursor()
            cursor.execute('''
                CREATE TABLE IF NOT EXISTS accounts (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    account_number TEXT UNIQUE NOT NULL,
                    owner_name TEXT NOT NULL,
                    balance REAL DEFAULT 0.0,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            ''')

            cursor.execute('''
                CREATE TABLE IF NOT EXISTS transactions (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    from_account TEXT,
                    to_account TEXT,
                    amount REAL NOT NULL,
                    type TEXT NOT NULL,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                )
            ''')

    def transfer_money(self, from_account, to_account, amount):
        """계좌 이체 (트랜잭션)"""
        if amount <= 0:
            raise ValueError("이체 금액은 0보다 커야 합니다")

        conn = sqlite3.connect(self.db_name)
        try:
            cursor = conn.cursor()

            # 출금 계좌 잔액 확인
            cursor.execute(
                "SELECT balance FROM accounts WHERE account_number = ?",
                (from_account,)
            )
            from_balance = cursor.fetchone()

            if not from_balance or from_balance[0] < amount:
                raise ValueError("잔액이 부족합니다")

            # 출금
            cursor.execute(
                "UPDATE accounts SET balance = balance - ? WHERE account_number = ?",
                (amount, from_account)
            )

            # 입금
            cursor.execute(
                "UPDATE accounts SET balance = balance + ? WHERE account_number = ?",
                (amount, to_account)
            )

            # 거래 기록
            cursor.execute(
                "INSERT INTO transactions (from_account, to_account, amount, type) VALUES (?, ?, ?, 'transfer')",
                (from_account, to_account, amount)
            )

            conn.commit()
            print(f"이체 성공: {from_account} → {to_account} ({amount}원)")

        except Exception as e:
            conn.rollback()
            print(f"이체 실패: {e}")
            raise

        finally:
            conn.close()

# 사용
acc_db = AccountDatabase()
acc_db.transfer_money("1001", "1002", 50000)
```

#### 대용량 CSV 파일 처리
```python
import csv
from typing import Iterator, Dict

class CSVProcessor:
    @staticmethod
    def read_large_csv(filename: str, chunk_size: int = 1000) -> Iterator[list]:
        """대용량 CSV를 청크 단위로 읽기"""
        chunk = []

        with open(filename, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)

            for row in reader:
                chunk.append(row)

                if len(chunk) >= chunk_size:
                    yield chunk
                    chunk = []

            if chunk:
                yield chunk

    @staticmethod
    def filter_csv(input_file: str, output_file: str, condition_func):
        """조건에 맞는 행만 필터링"""
        with open(input_file, 'r', encoding='utf-8') as fin, \
             open(output_file, 'w', encoding='utf-8', newline='') as fout:

            reader = csv.DictReader(fin)
            writer = csv.DictWriter(fout, fieldnames=reader.fieldnames)

            writer.writeheader()

            for row in reader:
                if condition_func(row):
                    writer.writerow(row)

    @staticmethod
    def aggregate_csv(filename: str, group_by: str, agg_col: str):
        """그룹별 집계"""
        from collections import defaultdict

        aggregated = defaultdict(float)

        with open(filename, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)

            for row in reader:
                key = row[group_by]
                value = float(row[agg_col])
                aggregated[key] += value

        return dict(aggregated)

# 사용 예제

# 1. 대용량 파일 처리
processor = CSVProcessor()
for chunk in processor.read_large_csv('large_data.csv', chunk_size=10000):
    # 청크별 처리
    print(f"처리 중: {len(chunk)}개 행")

# 2. 필터링 (나이 30 이상)
processor.filter_csv(
    'users.csv',
    'adults.csv',
    condition_func=lambda row: int(row.get('age', 0)) >= 30
)

# 3. 집계 (도시별 매출 합계)
totals = processor.aggregate_csv('sales.csv', group_by='city', agg_col='amount')
for city, total in totals.items():
    print(f"{city}: {total:,.0f}원")
```

```python
def csv_to_json(csv_file: str, json_file: str):
    """CSV를 JSON으로 변환"""

        json.dump(data, f, ensure_ascii=False, indent=2)
```

```python
class CSVCleaner:
    @staticmethod
    def clean_and_validate(input_file: str, output_file: str):
        """데이터 정제 및 검증"""
        errors = []
        cleaned_rows = []

        with open(input_file, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            fieldnames = reader.fieldnames

            for line_num, row in enumerate(reader, start=2):
                try:
                    # 공백 제거
                    row = {k: v.strip() if v else '' for k, v in row.items()}

                    # 필수 필드 검증
                    if not row.get('email'):
                        errors.append(f"라인 {line_num}: 이메일 누락")
                        continue

                    # 이메일 형식 검증
                    if '@' not in row['email']:
                        errors.append(f"라인 {line_num}: 잘못된 이메일 형식")
                        continue

                    # 숫자 필드 변환
                    if 'age' in row:
                        try:
                            age = int(row['age'])
                            if age < 0 or age > 150:
                                errors.append(f"라인 {line_num}: 나이 범위 오류")
                                continue
                        except ValueError:
                            errors.append(f"라인 {line_num}: 나이는 숫자여야 함")
                            continue

                    cleaned_rows.append(row)

                except Exception as e:
                    errors.append(f"라인 {line_num}: {str(e)}")

        # 정제된 데이터 저장
        if cleaned_rows:
            with open(output_file, 'w', encoding='utf-8', newline='') as f:
                writer = csv.DictWriter(f, fieldnames=fieldnames)
                writer.writeheader()
                writer.writerows(cleaned_rows)

        # 오류 리포트
        if errors:
            with open('errors.log', 'w', encoding='utf-8') as f:
                f.write('\n'.join(errors))

        return len(cleaned_rows), len(errors)

# 사용
cleaner = CSVCleaner()
cleaned, errors = cleaner.clean_and_validate('raw_data.csv', 'clean_data.csv')
print(f"정제 완료: {cleaned}개 행, {errors}개 오류")
```

```python
import requests
import json

class APIClient:
    def __init__(self, base_url):
        self.base_url = base_url

    def get_json(self, endpoint):
        """API에서 JSON 데이터 가져오기"""
        response = requests.get(f"{self.base_url}/{endpoint}")
        response.raise_for_status()
        return response.json()

        """JSON 데이터 전송"""

            f"{self.base_url}/{endpoint}",
            json=data,
            headers=headers
        )
        response.raise_for_status()
        return response.json()
```

```python
def validate_json_schema(data, schema):
    """JSON 스키마 검증"""

    def check_type(value, expected_type):
        type_map = {
            'string': str,
            'number': (int, float),
            'boolean': bool,
            'array': list,
            'object': dict,
            'null': type(None)
        }

        if expected_type not in type_map:
            return False

        return isinstance(value, type_map[expected_type])

    def validate_object(obj, obj_schema):
        if not isinstance(obj, dict):
            return False, "객체가 아님"

        # 필수 필드 확인
        required = obj_schema.get('required', [])
        for field in required:
            if field not in obj:
                return False, f"필수 필드 누락: {field}"

        # 각 필드 검증
        properties = obj_schema.get('properties', {})
        for key, value in obj.items():
            if key in properties:
                field_schema = properties[key]
                expected_type = field_schema.get('type')

                if not check_type(value, expected_type):
                    return False, f"필드 '{key}' 타입 오류"

        return True, "검증 성공"

    return validate_object(data, schema)

# 사용
user_schema = {
    'type': 'object',
    'required': ['name', 'email'],
    'properties': {
        'name': {'type': 'string'},
        'email': {'type': 'string'},
        'age': {'type': 'number'}
    }
}

user_data = {
    'name': '홍길동',
    'email': 'hong@example.com',
    'age': 30
}

is_valid, message = validate_json_schema(user_data, user_schema)
print(f"검증 결과: {message}")
```

```python
def get_nested_value(data, path, default=None):
    """
    중첩된 JSON에서 값 추출
    path 예: "user.address.city"
    """
    keys = path.split('.')
    value = data

    try:
        for key in keys:
            if isinstance(value, dict):
                value = value[key]
            elif isinstance(value, list):
                value = value[int(key)]
            else:
                return default
        return value
    except (KeyError, IndexError, ValueError):
        return default

# 사용
complex_data = {
    'user': {
        'name': '홍길동',
        'address': {
            'city': '서울',
            'zip': '12345'
        },
        'orders': [
            {'id': 1, 'total': 50000},
            {'id': 2, 'total': 30000}
        ]
    }
}

print(get_nested_value(complex_data, 'user.address.city'))  # 서울
print(get_nested_value(complex_data, 'user.orders.0.total'))  # 50000
print(get_nested_value(complex_data, 'user.phone', '없음'))  # 없음
```

```python
import xml.etree.ElementTree as ET
import requests
from datetime import datetime

class RSSFeedParser:
    def __init__(self, feed_url):
        self.feed_url = feed_url

    def fetch_and_parse(self):
        """RSS 피드 가져오기 및 파싱"""
        response = requests.get(self.feed_url)
        response.raise_for_status()

        root = ET.fromstring(response.content)
        items = []

        for item in root.findall('.//item'):
            article = {
                'title': item.find('title').text,
                'link': item.find('link').text,
                'description': item.find('description').text,
                'pubDate': item.find('pubDate').text
            }
            items.append(article)

        return items

    def get_latest(self, n=10):
        """최신 n개 아이템 가져오기"""
        items = self.fetch_and_parse()
        return items[:n]

# 사용
```

```python
import json

def xml_to_dict(element):
    """XML 요소를 딕셔너리로 변환"""
    result = {}

    # 속성 추가
    if element.attrib:
        result['@attributes'] = element.attrib

    # 자식 요소 처리
    children = list(element)

    if children:
        child_dict = {}
        for child in children:
            child_data = xml_to_dict(child)

            if child.tag in child_dict:
                # 이미 있으면 리스트로 변환
                if not isinstance(child_dict[child.tag], list):
                    child_dict[child.tag] = [child_dict[child.tag]]
                child_dict[child.tag].append(child_data)
            else:
                child_dict[child.tag] = child_data

        result.update(child_dict)

    # 텍스트 내용
    if element.text and element.text.strip():
        if children or element.attrib:
            result['#text'] = element.text.strip()
        else:
            return element.text.strip()

    return result

def xml_to_json(xml_file, json_file):
    """XML 파일을 JSON으로 변환"""
    tree = ET.parse(xml_file)
    root = tree.getroot()

    data = {root.tag: xml_to_dict(root)}

    with open(json_file, 'w', encoding='utf-8') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

    print(f"변환 완료: {xml_file} → {json_file}")

# 사용
xml_to_json('config.xml', 'config.json')
```

```python
import re
from collections import Counter
from datetime import datetime

class LogAnalyzer:
    def __init__(self, log_file):
        self.log_file = log_file
        self.logs = []

    def parse_apache_log(self):
        """Apache 로그 파싱"""
        pattern = r'(\S+) - - \[(.*?)\] "(.*?)" (\d+) (\d+)'

        with open(self.log_file, 'r') as f:
            for line in f:
                match = re.match(pattern, line)
                if match:
                    self.logs.append({
                        'ip': match.group(1),
                        'timestamp': match.group(2),
                        'request': match.group(3),
                        'status': int(match.group(4)),
                        'size': int(match.group(5))
                    })

        return len(self.logs)

    def get_top_ips(self, n=10):
        """상위 N개 IP"""
        ips = [log['ip'] for log in self.logs]
        return Counter(ips).most_common(n)

    def get_status_distribution(self):
        """HTTP 상태 코드 분포"""
        statuses = [log['status'] for log in self.logs]
        return dict(Counter(statuses))

    def get_error_logs(self):
        """에러 로그만 추출 (4xx, 5xx)"""
        return [log for log in self.logs if log['status'] >= 400]

# 사용
analyzer = LogAnalyzer('access.log')
analyzer.parse_apache_log()

print("상위 10개 IP:")
for ip, count in analyzer.get_top_ips(10):
    print(f"  {ip}: {count}회")

print("\n상태 코드 분포:")
for status, count in analyzer.get_status_distribution().items():
    print(f"  {status}: {count}회")

print(f"\n에러 로그: {len(analyzer.get_error_logs())}개")
```

```python
import time

class LogMonitor:
    def __init__(self, log_file, keywords=None):
        self.log_file = log_file
        self.keywords = keywords or ['ERROR', 'WARNING', 'CRITICAL']

    def tail_log(self, callback=None):
        """실시간 로그 모니터링"""
        with open(self.log_file, 'r') as f:
            # 파일 끝으로 이동
            f.seek(0, 2)

            while True:
                line = f.readline()

                if not line:
                    time.sleep(0.1)
                    continue

                # 키워드 체크
                for keyword in self.keywords:
                    if keyword in line:
                        if callback:
                            callback(line, keyword)
                        else:
                            print(f"[{keyword}] {line.strip()}")
```

```python
import os
import shutil
from pathlib import Path

class FileOrganizer:
    # 파일 확장자별 카테고리
    CATEGORIES = {
        '이미지': ['.jpg', '.jpeg', '.png', '.gif', '.bmp', '.svg'],
        '문서': ['.pdf', '.doc', '.docx', '.txt', '.xlsx', '.pptx'],
        '비디오': ['.mp4', '.avi', '.mov', '.mkv'],
        '음악': ['.mp3', '.wav', '.flac'],
        '압축': ['.zip', '.rar', '.7z', '.tar', '.gz'],
        '코드': ['.py', '.java', '.cpp', '.js', '.html', '.css']
    }

    def __init__(self, source_dir):
        self.source_dir = Path(source_dir)

    def organize_by_extension(self):
        """확장자별로 파일 정리"""
        if not self.source_dir.exists():
            print(f"디렉토리가 존재하지 않음: {self.source_dir}")
            return

        organized_count = 0

        for file_path in self.source_dir.glob('*'):
            if file_path.is_file():
                # 확장자로 카테고리 찾기
                ext = file_path.suffix.lower()
                category = self._get_category(ext)

                # 대상 디렉토리 생성
                target_dir = self.source_dir / category
                target_dir.mkdir(exist_ok=True)

                # 파일 이동
                target_path = target_dir / file_path.name
                shutil.move(str(file_path), str(target_path))

                organized_count += 1
                print(f"이동: {file_path.name} → {category}/")

        print(f"\n정리 완료: {organized_count}개 파일")

    def _get_category(self, ext):
        """확장자에 해당하는 카테고리 찾기"""
        for category, extensions in self.CATEGORIES.items():
            if ext in extensions:
                return category
        return '기타'

# 사용
organizer = FileOrganizer('/Users/username/Downloads')
organizer.organize_by_extension()
```

```python
from datetime import datetime

class DateBasedOrganizer:
    def __init__(self, source_dir):
        self.source_dir = Path(source_dir)

    def organize_by_date(self):
        """수정일 기준으로 날짜별 정리"""
        for file_path in self.source_dir.glob('*'):
            if file_path.is_file():
                # 파일 수정 시간
                mtime = os.path.getmtime(file_path)
                date = datetime.fromtimestamp(mtime)

                # YYYY/MM 형식으로 디렉토리 생성
                year_dir = self.source_dir / str(date.year)
                month_dir = year_dir / f"{date.month:02d}"
                month_dir.mkdir(parents=True, exist_ok=True)

                # 파일 이동
                target = month_dir / file_path.name
                shutil.move(str(file_path), str(target))

                print(f"이동: {file_path.name} → {date.year}/{date.month:02d}/")

# 사용
```

#### 중복 파일 찾기 및 제거
```python
import hashlib

class DuplicateFinder:
    def __init__(self, directory):
        self.directory = Path(directory)
        self.hashes = {}

    def find_duplicates(self):
        """중복 파일 찾기 (해시 기반)"""
        duplicates = []

        for file_path in self.directory.rglob('*'):
            if file_path.is_file():
                # 파일 해시 계산
                file_hash = self._get_file_hash(file_path)

                if file_hash in self.hashes:
                    # 중복 발견
                    duplicates.append((file_path, self.hashes[file_hash]))
                else:
                    self.hashes[file_hash] = file_path

        return duplicates

    def _get_file_hash(self, file_path):
        """파일의 MD5 해시 계산"""
        hasher = hashlib.md5()

        with open(file_path, 'rb') as f:
            while chunk := f.read(8192):
                hasher.update(chunk)

        return hasher.hexdigest()

    def remove_duplicates(self, dry_run=True):
        """중복 파일 제거"""
        duplicates = self.find_duplicates()

        if not duplicates:
            print("중복 파일 없음")
            return

        print(f"중복 파일 {len(duplicates)}개 발견:")

        removed = 0
        for duplicate, original in duplicates:
            print(f"  {duplicate.name} (원본: {original})")

            if not dry_run:
                duplicate.unlink()
                removed += 1

        if dry_run:
            print("\n[테스트 모드] 실제 삭제하려면 dry_run=False 설정")
        else:
            print(f"\n{removed}개 파일 삭제 완료")

# 사용
finder = DuplicateFinder('/path/to/directory')
finder.remove_duplicates(dry_run=True)  # 먼저 확인
# finder.remove_duplicates(dry_run=False)  # 실제 삭제
```
